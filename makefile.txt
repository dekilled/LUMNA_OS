# Orbit — wlroots 0.18 usa o módulo "wlroots".
# wlroots 0.19 muda o nome: make PKGS="wlroots-0.19 wayland-server xkbcommon"
PKGS ?= wlroots wayland-server xkbcommon
SRCS = server.c output.c view.c arrange.c input.c keybinds.c

orbit: $(SRCS) orbit.h
	$(CC) -g -Wall -Wextra -DWLR_USE_UNSTABLE -o $@ $(SRCS) $(shell pkg-config --cflags --libs $(PKGS))

clean:
	rm -f orbit

.PHONY: clean
