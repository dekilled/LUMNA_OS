# Navegador do LUMNA (CEF). Uso: cmake -S . -B build -G Ninja -DCEF_ROOT=/caminho/do/cef_binary_...  (veja dev.sh, que também monta a pasta de execução)
cmake_minimum_required(VERSION 3.19)
project(lumna_navegador C CXX)
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Passe -DCEF_ROOT=<pasta do cef_binary_...>")
endif()
set(CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
set(CMAKE_CXX_STANDARD 17)
find_package(CEF REQUIRED)
add_subdirectory(${CEF_ROOT}/libcef_dll libcef_dll_wrapper)
ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
add_executable(lumna-navegador src/main.cc src/midia.cc)
SET_EXECUTABLE_TARGET_PROPERTIES(lumna-navegador)
add_dependencies(lumna-navegador libcef_dll_wrapper)
target_link_libraries(lumna-navegador libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
